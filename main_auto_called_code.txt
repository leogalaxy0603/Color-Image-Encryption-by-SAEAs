for i = 1:initial_sample_size
    if NFEs >= MaxFEs, break; end
    f = FUN(hx(i,:));             % æ ‡é‡è¯„ä¼°
    hf(i) = f;
    NFEs = NFEs + 1;

    if f < gbestval
        gbestval = f; gbest = hx(i,:);
    end
    CE(NFEs) = gbestval;
end

% å»æ‰æœªè¯„ä¼°ï¼ˆinfï¼‰çš„è¡Œ
valid = isfinite(hf);
hx = hx(valid, :);
hf = hf(valid);

% åˆå§‹æ’åº
[hf, sidx] = sort(hf);
hx = hx(sidx, :);

% ---------------- æ¡†æ¶å‚æ•° ----------------
F  = 0.5;         % åŸºç¡€ç¼©æ”¾å› å­
CR = 0.5;         % äº¤å‰æ¦‚ç‡
gs = min( max(50, round(0.5*size(hx,1))) , size(hx,1) );    % è®­ç»ƒæ ·æœ¬æ•°ï¼ˆå…¨å±€ä»£ç†ï¼‰
ls = min(50, size(hx,1));                                   % å±€éƒ¨æ•°æ®æ¡æ•°

disp('====== Starting HSADE with Improved QUATRE (MaxFEs termination) ======');

% ================= ä¸»å¾ªç¯ï¼ˆæŒ‰é¢„ç®—é©±åŠ¨ï¼‰ =================
while NFEs < MaxFEs
    progress = NFEs / MaxFEs;

    % ---------- 1) å…¨å±€æœç´¢ï¼ˆåŸºäºä»£ç†çš„å€™é€‰ â†’ çœŸå®è¯„ä¼° 1 æ¬¡ï¼‰ ----------
    % é€‰å‡ºå½“å‰æœ€å¥½çš„å‰ NP ä¸ªä¸ªä½“ä½œä¸ºç§ç¾¤
    curNP = min(NP, size(hx,1));
    pop = hx(1:curNP, :);
    pop_fit = hf(1:curNP);

    VRmax = repmat(UB, curNP, 1);
    VRmin = repmat(LB, curNP, 1);

    % æ„å»ºå…¨å±€ä»£ç†ï¼ˆRBFï¼ŒMatlab çš„ newrbeï¼‰
    gs_use = min(gs, size(hx,1));
    ghx = hx(1:gs_use, :);
    ghf = hf(1:gs_use);
    ghxd = real(sqrt(ghx.^2*ones(size(ghx')) + ones(size(ghx))*(ghx').^2 - 2*ghx*(ghx')));
    spr = max(max(ghxd))/(D*gs_use)^(1/D);
    try
        net = newrbe(ghx', ghf, spr);
        modelFUN = @(x) sim(net, x')';  % è¿”å›è¡Œä¸º 1Ã—K
    catch
        % è‹¥ç¥ç»ç½‘ç»œå·¥å…·ç®±ä¸å¯ç”¨/å¤±è´¥ï¼Œåˆ™é€€åŒ–æˆç®€å•å¤šé¡¹å¼è¯„ä¼°å™¨ï¼ˆæç®€å…œåº•ï¼‰
        modelFUN = @(x) sum((x - mean(ghx,1)).^2, 2);
    end

    % å€™é€‰äº§ç”Ÿï¼ˆåªç”¨ä»£ç†æ’åºï¼Œä¸æ¶ˆè€—çœŸå®é¢„ç®—ï¼‰
    K  = 30;
    Uc = QUATRE_Improved(pop, curNP, D, F, CR, VRmax, VRmin, ...
                         'fitness', pop_fit', ...
                         'progress', progress, ...
                         'enable_adaptive', true);

    % ä»£ç†è¯„åˆ†å¹¶åšâ€œå€™é€‰èšåˆâ€
    pred = modelFUN(Uc);                 % 1Ã—curNP æˆ– curNPÃ—1
    [~, ord] = sort(pred(:), 'ascend');
    chooseK = min(K, size(Uc,1));
    pool = Uc(ord(1:chooseK), :);

    % éšæœºå–å‰è‹¥å¹²å†å–å‡å€¼ï¼ˆå¹³è¡¡æ¢ç´¢/å¼€å‘ï¼‰
    rr = randi(chooseK); 
    col = pool(1:rr, :);
    cand = mean(col, 1);

    % çœŸå®è¯„ä¼°ï¼ˆæ¶ˆè€—é¢„ç®— 1 æ¬¡ï¼‰
    if NFEs >= MaxFEs, break; end
    cand_fit = FUN(cand);
    NFEs = NFEs + 1;

    % æ›´æ–°æ•°æ®åº“
    hx = [hx; cand];
    hf = [hf, cand_fit];
    [hf, sidx] = sort(hf);
    hx = hx(sidx, :);

    % æ›´æ–°å…¨å±€æœ€å¥½
    if cand_fit <= gbestval
        gbestval = cand_fit; gbest = cand;
        GS_sucNum = GS_sucNum + 1;
        disp(['Best Cost(GS) = ' num2str(gbestval, '%.6e') ' | NFE=' num2str(NFEs) ' | prog=' num2str(progress, '%.2f')]);
    end

    CE(NFEs) = gbestval;
    if NFEs >= MaxFEs, break; end

    % ---------- 2) å±€éƒ¨æœç´¢ï¼ˆå¸¦é¢„ç®—çš„æ”¹è¿› LPSRï¼‰ ----------
    % å–å‰ ls æ¡ä½œä¸ºå±€éƒ¨æ•°æ®
    ls_use = min(ls, size(hx,1));
    lhx = hx(1:ls_use, :);
    lhf = hf(1:ls_use);

    local_LB = min(lhx, [], 1);
    local_UB = max(lhx, [], 1);

    % å±€éƒ¨ä»£ç†
    lhxd = real(sqrt(lhx.^2*ones(size(lhx')) + ones(size(lhx))*(lhx').^2 - 2*lhx*(lhx')));
    sprL = max(max(lhxd))/(D*ls_use)^(1/D);
    try
        netL = newrbe(lhx', lhf, sprL);
        LocalModelFUN = @(x) sim(netL, x')';
    catch
        LocalModelFUN = @(x) sum((x - mean(lhx,1)).^2, 2);
    end

    % ä¼ å…¥â€œå‰©ä½™é¢„ç®—â€
    remainBudget = MaxFEs - NFEs;
    if remainBudget <= 0, break; end

    maxLocalFEs = min(remainBudget, 50*D);   % ç»™å±€éƒ¨æœç´¢ä¸€ä¸ª cap
    minerror = 1e-8;

    [candLS, candLS_fit, usedFEs] = LocalSearch_LPSR_Improved( ...
        D, maxLocalFEs, FUN, minerror, lhx, local_LB, local_UB, LocalModelFUN);

    NFEs = NFEs + usedFEs;
    if usedFEs > 0
        % æŠŠ candLS æ’åº“ï¼ˆcandLS_fit å·²ç»æ˜¯çœŸå®è¯„ä¼°ï¼‰
        hx = [hx; candLS];
        hf = [hf, candLS_fit];
        [hf, sidx] = sort(hf);
        hx = hx(sidx, :);

        if candLS_fit <= gbestval
            gbestval = candLS_fit; gbest = candLS;
            LS_sucNum = LS_sucNum + 1;
            disp(['Best Cost(LS) = ' num2str(gbestval, '%.6e') ' | NFE=' num2str(NFEs)]);
        end
        CE(NFEs) = gbestval;
    end
end

SN = [GS_sucNum, LS_sucNum];
disp('====== Optimization Complete ======');
disp(['Final best value: ', num2str(gbestval, '%.6e')]);
disp(['Total NFEs used : ', num2str(NFEs)]);
% æˆªæ–­ CE åˆ°å®é™… NFEs é•¿åº¦
CE = CE(1:NFEs);
end


% ================= æ”¹è¿›çš„ QUATRE ç”Ÿæˆç®—å­ï¼ˆä¸åšçœŸå®è¯„ä¼°ï¼‰ =================
function U = QUATRE_Improved(P, NP, Dim, F, CR, VRmax, VRmin, varargin)
    p = inputParser;
    addParameter(p, 'fitness', [], @isnumeric);
    addParameter(p, 'progress', 0.5, @(x) x>=0 && x<=1);
    addParameter(p, 'enable_adaptive', true, @islogical);
    parse(p, varargin{:});

    fitness = p.Results.fitness;
    progress = p.Results.progress;
    enable_adaptive = p.Results.enable_adaptive;

    if isempty(fitness)
        fitness = sum(P.^2, 2); % ç®€å•å…œåº•
    end
    [~, gbestid] = min(fitness);
    gbest = P(gbestid, :);

    % ç”Ÿæˆä¸‹ä¸‰è§’æ©ç  label
    K = floor(NP/Dim);
    label = zeros(NP, Dim);
    for j = 0:K-1
        label(j*Dim+1:j*Dim+Dim, :) = tril(ones(Dim, Dim), 0);
    end
    if K*Dim < NP
        label(K*Dim+1:NP, :) = tril(ones(NP-K*Dim, Dim), 0);
    end
    for t = 1:NP
        label(t, :) = label(t, randperm(Dim));
    end
    label = label(randperm(NP)', :);

    % è‡ªé€‚åº” F
    F_final = F;
    if enable_adaptive
        diversity = mean(std(P, 0, 1));
        max_range = mean(VRmax(1,:) - VRmin(1,:));
        diversity_ratio = max(1e-12, diversity / max_range);
        F_adp = F * (0.5 + diversity_ratio);
        F_adp = max(0.3, min(1.0, F_adp));
        F_prog = F * (1.2 - 0.5*progress);
        F_final = 0.7*F_adp + 0.3*F_prog;
    end

    U = zeros(NP, Dim);
    for i = 1:NP
        % ç­–ç•¥é€‰æ‹©
        if enable_adaptive
            r = rand();
            if progress < 0.3
                strat = (r < 0.6) + 1;       % 1: rand/1, 2: current-to-best/1
            elseif progress < 0.7
                if r < 0.3, strat = 1; elseif r < 0.7, strat = 2; else, strat = 3; end
            else
                strat = 2 + (r >= 0.2);      % 2: current-to-best/1, 3: best/1
            end
        else
            strat = 3;
        end

        % é€‰ r1,r2,r3
        idx = randperm(NP); idx(idx==i) = [];
        k0 = idx(1); k1 = idx(2); k2 = idx(3);
        P1 = P(k0,:); P2 = P(k1,:); P3 = P(k2,:);

        switch strat
            case 1 % rand/1
                V = P1 + F_final*(P2 - P3);
            case 2 % current-to-best/1
                V = P(i,:) + F_final*(gbest - P(i,:)) + F_final*(P2 - P3);
            otherwise % best/1
                V = gbest + F_final*(P2 - P3);
        end

        % QUATRE æ©ç æ‹·è´
        Ui = P(i,:);
        Ui(logical(label(i,:))) = V(logical(label(i,:)));

        % è¾¹ç•Œå¤„ç†
        Ui = handleBoundary(Ui, P(i,:), VRmin(i,:), VRmax(i,:));

        U(i,:) = Ui;
    end
end

% â€”â€” è¾¹ç•Œå¤„ç†ï¼šé•œåƒ + ä¸­ç‚¹ç­–ç•¥ + è£å‰ª â€”â€” 
function x_new = handleBoundary(x, x_old, LB, UB)
    x_new = x;
    for j = 1:length(x)
        if x(j) < LB(j) || x(j) > UB(j)
            if rand() < 0.8
                if x(j) < LB(j)
                    x_new(j) = LB(j) + (LB(j) - x(j));
                    if x_new(j) > UB(j), x_new(j) = (LB(j)+UB(j))/2; end
                else
                    x_new(j) = UB(j) - (x(j) - UB(j));
                    if x_new(j) < LB(j), x_new(j) = (LB(j)+UB(j))/2; end
                end
            else
                if x(j) < LB(j), x_new(j) = (LB(j)+x_old(j))/2;
                else,            x_new(j) = (UB(j)+x_old(j))/2; end
            end
            x_new(j) = max(LB(j), min(UB(j), x_new(j)));
        end
    end
end

% ================== å±€éƒ¨æœç´¢ï¼ˆå¸¦é¢„ç®—ã€é€ç‚¹è¯„ä¼°ç‰ˆï¼‰ ==================
function [bestX, bestFit, usedFEs] = LocalSearch_LPSR_Improved( ...
    Dim, maxFes, FUN, minerror, lhx, LB, UB, LocalModelFUN)

    NP0 = 50; max_NP = NP0; min_NP = 4;
    F = 0.5; CR = 0.8;

    % åˆå§‹åŒ–
    NP = NP0;
    X  = (UB - LB).*rand(NP,Dim) + LB;

    % çœŸå®è¯„ä¼°ï¼ˆå°½é‡è¯„å®Œåˆå§‹ç§ç¾¤ï¼‰
    fitnessX = inf(NP,1);
    evalX    = false(NP,1);      % æ˜¯å¦çœŸå®è¯„ä¼°æ ‡è®°
    usedFEs  = 0;
    for i = 1:NP
        if usedFEs >= maxFes, break; end
        fitnessX(i) = FUN(X(i,:));
        evalX(i)    = true;
        usedFEs     = usedFEs + 1;
    end
    % æœªè¯„ä¼°çš„ä»…ç”¨äºâ€œå ä½â€ï¼Œä¸ç»™æœ€ä¼˜æ¯”è¾ƒ
    if any(~evalX)
        fitnessX(~evalX) = LocalModelFUN(X(~evalX,:));
    end

    % åªåœ¨çœŸå®è¯„ä¼°è¿‡çš„é‡Œæ‰¾æœ€ä¼˜ï¼›è‹¥ä¸€ä¸ªéƒ½æ²¡è¯„ä¼°ï¼Œåˆ™é€€å‡º
    if any(evalX)
        [bestFit, idxbest] = min(fitnessX(evalX));
        idxList = find(evalX);
        bestX   = X(idxList(idxbest),:);
    else
        bestFit = inf;
        bestX   = X(1,:);
        return;
    end

    emaImp = 0; beta = 0.8; tau_imp = 1e-5;

    while usedFEs < maxFes
        fitnessBest_old = bestFit;

        % å˜å¼‚ + äº¤å‰
        V = zeros(NP,Dim); U = zeros(NP,Dim);
        [~, sorted_index] = sort(fitnessX);     % æ— ç¢ï¼šåªæ˜¯ç”¨äºé€‰ pbest
        pNP = max(round(0.2*NP),2);

        for i = 1:NP
            pbest_idx = sorted_index(randi(pNP));
            pbest = X(pbest_idx,:);

            r1 = randi(NP); while r1==i, r1 = randi(NP); end
            r2 = randi(NP); while r2==i || r2==r1, r2 = randi(NP); end

            V(i,:) = X(i,:) + F*(pbest - X(i,:)) + F*(X(r1,:) - X(r2,:));
            V(i,:) = max(LB, min(UB, V(i,:)));
        end

        for i = 1:NP
            jrand = randi(Dim);
            for j = 1:Dim
                if rand <= CR || j == jrand
                    U(i,j) = V(i,j);
                else
                    U(i,j) = X(i,j);
                end
            end
        end

        % â€”â€” é€ç‚¹çœŸå®è¯„ä¼°ï¼šåªè¦é¢„ç®—å…è®¸å°±è¯„ï¼Œå¦åˆ™æ ‡ä¸ºâ€œæœªè¯„ä¼°â€ â€”â€” 
        fitnessU = inf(NP,1);
        evalU    = false(NP,1);
        for i = 1:NP
            if usedFEs >= maxFes, break; end
            fitnessU(i) = FUN(U(i,:));
            evalU(i)    = true;
            usedFEs     = usedFEs + 1;
        end
        % æœªè¯„ä¼°é¡¹ç”¨ä»£ç†ä½œå ä½ï¼ˆä»…ç”¨äºæ’åº/å¯†åº¦åˆ¤æ–­ï¼‰ï¼Œ**ä¸å‚ä¸**æ›¿æ¢ä¸æœ€ä¼˜æ¯”è¾ƒ
        if any(~evalU)
            fitnessU(~evalU) = LocalModelFUN(U(~evalU,:));
        end

        % â€”â€” 1 å¯¹ 1 é€‰æ‹©ï¼šåªæœ‰ evalU(i)=true æ—¶æ‰å…è®¸æ›¿æ¢ â€”â€” 
        for i = 1:NP
            if evalU(i) && (fitnessU(i) <= fitnessX(i))
                X(i,:)       = U(i,:);
                fitnessX(i)  = fitnessU(i);
                evalX(i)     = true;      % è¯¥ä¸ªä½“æœ‰çœŸå®è¯„ä¼°
                if fitnessU(i) < bestFit
                    bestFit = fitnessU(i);
                    bestX   = X(i,:);
                end
            end
        end

        % EMA æ”¹å–„ç‡
        rel_imp = max(0, (fitnessBest_old - bestFit) / (abs(fitnessBest_old) + 1e-12));
        emaImp  = beta*emaImp + (1-beta)*rel_imp;

        NP_linear = round(((min_NP - max_NP)/maxFes)*usedFEs + max_NP);
        next_NP = NP;
        if emaImp < tau_imp
            next_NP = min(NP, NP_linear);
        end

        % æ”¶ç¼©ï¼ˆå¯†åº¦ + æœ€å·®ï¼‰ï¼Œå…è®¸ä½¿ç”¨â€œå ä½ fitnessâ€åšå¯†åº¦/æ’åºï¼Œä½†ä¸æ”¹æœ€ä¼˜
        if next_NP < NP
            re_num = NP - next_NP;
            if next_NP < min_NP, re_num = NP - min_NP; end

            XX = sum(X.^2,2);
            dmat = sqrt(max(0, XX + XX' - 2*(X*X')));
            dmat(1:NP+1:end) = inf;
            nnDist = min(dmat, [], 2);
            [~, densIdx]  = sort(nnDist,'ascend');
            [~, worstIdx] = sort(fitnessX,'descend');

            take_dens  = floor(re_num/2);
            take_worst = re_num - take_dens;
            delIdx = unique([densIdx(1:take_dens); worstIdx(1:take_worst)]);

            k = 1;
            while numel(delIdx) < re_num && k <= NP
                c = worstIdx(k);
                if ~ismember(c, delIdx), delIdx(end+1,1) = c; end
                k = k + 1;
            end

            X(delIdx,:)      = [];
            fitnessX(delIdx) = [];
            evalX(delIdx)    = [];
            NP = size(X,1);
        end

        if abs(fitnessBest_old - bestFit) <= minerror
            break;
        end
    end
end

% ====================== FILE: DNA_yunsuan.m ======================

%% ×Óº¯Êı DNAÔËËã arr1ºÍarr2±íÊ¾Á½¸öÒªÔËËãµÄDNA¾ØÕó¡£
%numÎª0Ê±£¬±íÊ¾¼Ó£»numÎª1Ê±£¬±íÊ¾¼õ£»numÎª2Ê±£¬±íÊ¾Òì»ò£»numÎª3Ê±±íÊ¾Í¬»ò
function fv=DNA_yunsuan(arr1,arr2,num)
[m,n]=size(arr1);
fv=zeros(m,n);          %Ô¤·ÖÅäÄÚ´æ
if num==0    %¼Ó·¨
    for i=1:m
        for j=1:n
            if arr1(i,j)=='A'
                fv(i,j)=arr2(i,j);
            elseif arr1(i,j)=='T'
                if arr2(i,j)=='A'
                    fv(i,j)='T';
                elseif arr2(i,j)=='T'
                    fv(i,j)='C';
                elseif arr2(i,j)=='C'
                    fv(i,j)='G';
                else
                    fv(i,j)='A';
                end
            elseif arr1(i,j)=='C'
                if arr2(i,j)=='A'
                    fv(i,j)='C';
                elseif arr2(i,j)=='T'
                    fv(i,j)='G';
                elseif arr2(i,j)=='C'
                    fv(i,j)='A';
                else
                    fv(i,j)='T';
                end
            else
                if arr2(i,j)=='A'
                    fv(i,j)='G';
                elseif arr2(i,j)=='T'
                    fv(i,j)='A';
                elseif arr2(i,j)=='C'
                    fv(i,j)='T';
                else
                    fv(i,j)='C';
                end
            end
        end
    end
elseif num==1       %¼õ·¨
    for i=1:m
        for j=1:n
            if arr2(i,j)=='A'
                fv(i,j)=arr1(i,j);
            elseif arr2(i,j)=='T'
                if arr1(i,j)=='A'
                    fv(i,j)='G';
                elseif arr1(i,j)=='T'
                    fv(i,j)='A';
                elseif arr1(i,j)=='C'
                    fv(i,j)='T';
                else
                    fv(i,j)='C';
                end
            elseif arr2(i,j)=='C'
                if arr1(i,j)=='A'
                    fv(i,j)='C';
                elseif arr1(i,j)=='T'
                    fv(i,j)='G';
                elseif arr1(i,j)=='C'
                    fv(i,j)='A';
                else
                    fv(i,j)='T';
                end
            else
                if arr1(i,j)=='A'
                    fv(i,j)='T';
                elseif arr1(i,j)=='T'
                    fv(i,j)='C';
                elseif arr1(i,j)=='C'
                    fv(i,j)='G';
                else
                    fv(i,j)='A';
                end
            end
        end
    end
elseif num==2        %Òì»ò
    for i=1:m
        for j=1:n
            if arr1(i,j)==arr2(i,j)
                fv(i,j)='C';
            elseif (arr1(i,j)=='T' && arr2(i,j)=='A') || (arr1(i,j)=='A' && arr2(i,j)=='T') || (arr1(i,j)=='G' && arr2(i,j)=='C') || (arr1(i,j)=='C' && arr2(i,j)=='G')
                fv(i,j)='G';
            elseif (arr1(i,j)=='C' && arr2(i,j)=='A') || (arr1(i,j)=='A' && arr2(i,j)=='C') || (arr1(i,j)=='G' && arr2(i,j)=='T') || (arr1(i,j)=='T' && arr2(i,j)=='G')
                fv(i,j)='A';
            else
                fv(i,j)='T';
            end
        end
    end
else                %Í¬»ò
    for i=1:m
        for j=1:n
            if arr1(i,j)==arr2(i,j)
                fv(i,j)='A';
            elseif (arr1(i,j)=='T' && arr2(i,j)=='A') || (arr1(i,j)=='A' && arr2(i,j)=='T') || (arr1(i,j)=='G' && arr2(i,j)=='C') || (arr1(i,j)=='C' && arr2(i,j)=='G')
                fv(i,j)='T';
            elseif (arr1(i,j)=='C' && arr2(i,j)=='A') || (arr1(i,j)=='A' && arr2(i,j)=='C') || (arr1(i,j)=='G' && arr2(i,j)=='T') || (arr1(i,j)=='T' && arr2(i,j)=='G')
                fv(i,j)='C';
            else
                fv(i,j)='G';
            end
        end
    end
end


% ====================== FILE: DNA_jie.m ======================

%% ×Óº¯Êı DNA½âÂë
function fv=DNA_jie(array,num)
[m,n]=size(array);
A=zeros(m,n);           %Ô¤·ÖÅäÄÚ´æ
if num==1 || num==2
    for i=1:m
        for j=1:n
            if array(i,j)=='A'
                A(i,j)=0;
            elseif array(i,j)=='T'
                A(i,j)=3;
            elseif array(i,j)=='G'
                if num==1
                    A(i,j)=1;
                else
                    A(i,j)=2;
                end
            else
                if num==1
                    A(i,j)=2;
                else
                    A(i,j)=1;
                end
            end
        end
    end
elseif num==3 || num==4
    for i=1:m
        for j=1:n
            if array(i,j)=='A'
                A(i,j)=1;
            elseif array(i,j)=='T'
                A(i,j)=2;
            elseif array(i,j)=='G'
                if num==3
                    A(i,j)=0;
                else
                    A(i,j)=3;
                end
            else
                if num==3
                    A(i,j)=3;
                else
                    A(i,j)=0;
                end
            end
        end
    end
elseif num==5 || num==6
    for i=1:m
        for j=1:n
            if array(i,j)=='A'
                A(i,j)=2;
            elseif array(i,j)=='T'
                A(i,j)=1;
            elseif array(i,j)=='G'
                if num==5
                    A(i,j)=0;
                else
                    A(i,j)=3;
                end
            else
                if num==5
                    A(i,j)=3;
                else
                    A(i,j)=0;
                end
            end
        end
    end
else                    %  num==7 || num==8
    for i=1:m
        for j=1:n
            if array(i,j)=='A'
                A(i,j)=3;
            elseif array(i,j)=='T'
                A(i,j)=0;
            elseif array(i,j)=='G'
                if num==7
                    A(i,j)=1;
                else
                    A(i,j)=2;
                end
            else
                if num==7
                    A(i,j)=2;
                else
                    A(i,j)=1;
                end
            end
        end
    end
end
A1=A(:,1:m);
A2=A(:,m+1:2*m);
A3=A(:,2*m+1:3*m);
A4=A(:,3*m+1:4*m);
fv=A1*64+A2*16+A3*4+A4;
end

% ====================== FILE: After_information_entropy.m ======================

function [xxs2_R,xxs2_G,xxs2_B]=After_information_entropy(Q_R,Q_G,Q_B)
T2_R=imhist(Q_R);
S2_R=sum(T2_R);

%GÍ¨µÀ

S2_G=sum(T2_G);
xxs2_G=0;
%BÍ¨µÀ
T2_B=imhist(Q_B);
S2_B=sum(T2_B);
xxs2_B=0;

    pp2_R=T2_R(i)/S2_R;
    pp2_G=T2_G(i)/S2_G;
    pp2_B=T2_B(i)/S2_B;
    if pp2_R~=0
        xxs2_R=xxs2_R-pp2_R*log2(pp2_R);
    end
    if pp2_G~=0
        xxs2_G=xxs2_G-pp2_G*log2(pp2_G);
    end
    if pp2_B~=0
        xxs2_B=xxs2_B-pp2_B*log2(pp2_B);
    end
end
end

function fitness=fobj(RXY2_R,RXY2_G,RXY2_B)
% %RXY2_R(1,1)=RXY2_SP_R;RXY2_R(1,2)=RXY2_CZ_R;RXY2_R(1,3)=RXY2_DJX_R;
% RXY2_G(2,1)=RXY2_SP_G;RXY2_G(2,2)=RXY2_CZ_G;RXY2_G(2,3)=RXY2_DJX_G;
% RXY2_B(3,1)=RXY2_SP_B;RXY2_B(3,2)=RXY2_CZ_B;RXY2_B(3,3)=RXY2_DJX_B;
RXY2_SP_R=RXY2_R(1,1);RXY2_CZ_R=RXY2_R(1,2);RXY2_DJX_R=RXY2_R(1,3);

fitness=abs(fitness_R+fitness_G+fitness_B);
end
